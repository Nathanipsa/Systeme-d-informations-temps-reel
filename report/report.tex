\begin{tcblisting}{title={Retriving the WCET, Q1, Q2 and Q3 of the task $\tau_1$},
pythonexample={Coucouibou}}
time_1 = 0
WCET = 0
times = []

security_factor = 1.3


for i in tqdm(range(10000), desc="Calculating WCET"):
    start_time = time.time()
    subprocess.run(["make", "-f", "MakeFile"], stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL)
    end_time = time.time()

    time_1 = end_time - start_time
    
    if time_1 > WCET:
        WCET = time_1

    times.append(time_1)
    Q1 = sorted(times)[int(0.25 * len(times))]
    Q3 = sorted(times)[int(0.75 * len(times))]
    Q2 = sorted(times)[int(0.5 * len(times))]


print("Worst Case Execution Time (WCET) : ", WCET)
print("Secure Execution Time (SET) : ", WCET * security_factor)
print("Q1 : ", Q1)
print("Q2 : ", Q2)
print("Q3 : ", Q3)
\end{tcblisting}
